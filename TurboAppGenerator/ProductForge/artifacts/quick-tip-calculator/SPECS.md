- React, Vue, Angular add 40-100KB overhead

- Single HTML file deployment doesn't require bundling
- Tailwind CSS compiled at build time (dev-only step)
- No module resolution needed at runtime
- Simpler deployment pipeline

### 2. Calculation Data Model (In-Memory, Transient)

**File:** `src/modules/calculator.js`

```javascript
/**
 * CALCULATION_RESULT
 * 
 * Immutable result object returned by calculateTip().
 * Contains both raw (full precision) and rounded values for audit trail.
 * 
 * TypeScript equivalent:
 * interface CalculationResult {
 *   billAmount: number;
 *   tipPercentage: number;
 *   tipAmountRaw: number;
 *   tipAmountRounded: number;
 *   totalRaw: number;
 *   totalRounded: number;
 *   currencyCode: string;
 *   decimals: number;
 *   timestamp: number;
 * }
 */

export function calculateTip(billAmount, tipPercentage, currencyCode) {
  // Step 1: Validate inputs
  if (!isValidBillAmount(billAmount)) {
    throw new ValidationError(`Invalid bill amount: ${billAmount}`);
  }
  if (!isValidTipPercentage(tipPercentage)) {
    throw new ValidationError(`Invalid tip percentage: ${tipPercentage}`);
  }
  if (!isValidCurrencyCode(currencyCode)) {
    throw new ValidationError(`Invalid currency code: ${currencyCode}`);
  }
  
  // Step 2: Get currency metadata
  const currencyMeta = getCurrencyMetadata(currencyCode);
  const decimals = currencyMeta.decimals;
  
  // Step 3: Calculate with FULL PRECISION (no rounding yet)
  // Keep at least 10 decimal places for intermediate calculations
  const tipAmountRaw = billAmount * (tipPercentage / 100);
  const totalRaw = billAmount + tipAmountRaw;
  
  // Step 4: Apply banker's rounding ONLY at display time
  const tipAmountRounded = bankerRound(tipAmountRaw, decimals);
  const totalRounded = bankerRound(totalRaw, decimals);
  
  // Step 5: Return immutable result object
  return Object.freeze({
    billAmount,
    tipPercentage,
    tipAmountRaw,           // Full precision (for backend validation if needed)
    tipAmountRounded,       // Rounded to currency decimals (for display)
    totalRaw,               // Full precision (for backend validation if needed)
    totalRounded,           // Rounded to currency decimals (for display)
    currencyCode,
    decimals,
    timestamp: Date.now()
  });
}

/**
 * BANKER'S ROUNDING ALGORITHM
 * 
 * Round-half-to-even: When exactly halfway between two values,
 * round to the nearest even number.
 * 
 * Examples:
 * - 2.5 → 2 (even)
 * - 3.5 → 4 (even)
 * - 2.45 → 2.4 (not exactly halfway)
 * - 2.55 → 2.6 (not exactly halfway due to floating-point)
 * 
 * Why banker's rounding?
 * - Minimizes systematic bias in financial calculations
 * - Standard in banking and accounting (ISO 80000-2)
 * - Reduces cumulative rounding errors in large datasets
 */
export function bankerRound(value, decimals) {
  // Validate inputs
  if (typeof value !== 'number' || !Number.isFinite(value)) {
    throw new Error(`Invalid value for rounding: ${value}`);
  }
  if (!Number.isInteger(decimals) || decimals < 0) {
    throw new Error(`Invalid decimals: ${decimals}`);
  }
  
  // Special case: no rounding needed
  if (decimals === 0) {
    const shifted = value;
    const floor = Math.floor(shifted);
    const remainder = shifted - floor;
    
    if (remainder < 0.5) return floor;
    if (remainder > 0.5) return floor + 1;
    // remainder === 0.5: round to nearest even
    return floor % 2 === 0 ? floor : floor + 1;
  }
  
  // General case: round to N decimal places
  const factor = Math.pow(10, decimals);
  const shifted = value * factor;
  const floor = Math.floor(shifted);
  const remainder = shifted - floor;
  
  // Tolerance for floating-point comparison (1e-10)
  const EPSILON = 1e-10;
  
  if (remainder < 0.5 - EPSILON) {
    return floor / factor;
  }
  if (remainder > 0.5 + EPSILON) {
    return (floor + 1) / factor;
  }
  
  // remainder ≈ 0.5: round to nearest even
  const rounded = floor % 2 === 0 ? floor : floor + 1;
  return rounded / factor;
}

/**
 * Validation helpers
 */
export function isValidBillAmount(amount) {
  if (typeof amount !== 'number' || !Number.isFinite(amount)) return false;
  if (amount < 0.01 || amount > 999999.99) return false;
  // Check decimal places (max 2)
  const decimalPlaces = (amount.toString().split('.')[1] || '').length;
  return decimalPlaces <= 2;
}

export function isValidTipPercentage(percentage) {
  if (typeof percentage !== 'number' || !Number.isFinite(percentage)) return false;
  if (percentage < 0 || percentage > 100) return false;
  // Check decimal places (max 1)
  const decimalPlaces = (percentage.toString().split('.')[1] || '').length;
  return decimalPlaces <= 1;
}

export function isValidCurrencyCode(code) {
  return code in CURRENCY_METADATA;
}
```

### 3. Input Validation Data Model

**File:** `src/modules/validator.js`

```javascript
/**
 * VALIDATION_RESULT
 * 
 * Immutable result object returned by validation functions.
 * 
 * TypeScript equivalent:
 * interface ValidationResult {
 *   isValid: boolean;
 *   value: number | null;
 *   error: string | null;
 *   fieldName: string;
 * }
 */

export function validateBillAmount(input) {
  // Input: string from HTML input element
  // Output: ValidationResult
  
  const fieldName = 'billAmount';
  
  // Step 1: Check if empty
  if (input === '' || input === null || input === undefined) {
    return {
      isValid: false,
      value: null,
      error: 'Bill amount is required',
      fieldName
    };
  }
  
  // Step 2: Parse to number
  const parsed = parseFloat(input);
  if (isNaN(parsed)) {
    return {
      isValid: false,
      value: null,
      error: 'Bill amount must be a number',
      fieldName
    };
  }
  
  // Step 3: Check range
  if (parsed < 0.01) {
    return {
      isValid: false,
      value: null,
      error: 'Bill amount must be at least $0.01',
      fieldName
    };
  }
  if (parsed > 999999.99) {
    return {
      isValid: false,
      value: null,
      error: 'Bill amount cannot exceed $999,999.99',
      fieldName
    };
  }
  
  // Step 4: Check decimal places (max 2)
  const decimalPlaces = (input.split('.')[1] || '').length;
  if (decimalPlaces > 2) {
    return {
      isValid: false,
      value: null,
      error: 'Bill amount can have at most 2 decimal places',
      fieldName
    };
  }
  
  // Step 5: Success
  return {
    isValid: true,
    value: parsed,
    error: null,
    fieldName
  };
}
```

```javascript
/**
 * LOCALSTORAGE_SCHEMA_V1
 * 
 * Persistent user preferences stored in browser localStorage.
 * 
 * Key: 'tipCalculator_prefs_v1'
 * Value: JSON string containing user preferences
 * 
 * TypeScript equivalent:
 * interface StorageSchema {
 *   selectedCurrency: string;
 *   lastUpdated: number;
 *   schemaVersion: number;
 * }
 */

const STORAGE_KEY = 'tipCalculator_prefs_v1';
const SCHEMA_VERSION = 1;
const DEFAULT_CURRENCY = 'USD';

/**
 * STORAGE_SCHEMA_V1
 * 
 * Exact structure of data stored in localStorage.
 * Must be JSON-serializable.
 */
const STORAGE_SCHEMA_V1 = {
  selectedCurrency: {
    type: 'string',
    required: true,
    enum: ['USD', 'EUR', 'GBP', 'JPY', 'CAD', 'AUD', 'CHF', 'INR', 'MXN', 'SGD'],
    default: 'USD',
    description: 'User-selected currency code'
  },
  lastUpdated: {
    type: 'number',
    required: true,
    format: 'unix_timestamp_ms',
    description: 'ISO 8601 timestamp (milliseconds since epoch) when preferences were last updated'
  },
  schemaVersion: {
    type: 'number',
    required: true,
    value: 1,
    description: 'Schema version for future migrations'
  }
};

/**
 * Load preferences from localStorage
 * 
 * Returns: { selectedCurrency: string, lastUpdated: number, schemaVersion: number }
 * Throws: Error if localStorage is unavailable or data is corrupted
 */
export function loadPreferences() {
  try {
    // Check if localStorage is available
    if (!isLocalStorageAvailable()) {
      console.warn('localStorage not available; using defaults');
      return getDefaultPreferences();
    }
    
    // Retrieve stored data
    const stored = localStorage.getItem(STORAGE_KEY);
    if (!stored) {
      // No stored preferences; return defaults
      return getDefaultPreferences();
    }
    
    // Parse JSON
    let parsed;
    try {
      parsed = JSON.parse(stored);
    } catch (e) {
      console.error('Failed to parse stored preferences; using defaults', e);
      return getDefaultPreferences();
    }
    
    // Validate schema
    if (parsed.schemaVersion !== SCHEMA_VERSION) {
      console.warn(`Schema version mismatch (expected ${SCHEMA_VERSION}, got ${parsed.schemaVersion}); using defaults`);
      return getDefaultPreferences();
    }
    
    // Validate selectedCurrency
    if (!isValidCurrencyCode(parsed.selectedCurrency)) {
```

**Banker's Rounding Algorithm (EXACT IMPLEMENTATION):**

```javascript
/**
 * Banker's rounding (round-half-to-even).
 * Minimizes systematic bias in financial calculations.
 * 
 * @param {number} value - Value to round
 * @param {number} decimals - Number of decimal places (0 or 2)
 * @returns {number} Rounded value
 * 
 * Algorithm:
 * 1. Shift value by 10^decimals (e.g., 47.505 * 100 = 4750.5)
 * 2. Separate into integer and fractional parts
 * 3. If fractional part < 0.5: round down
 * 4. If fractional part > 0.5: round up
 * 5. If fractional part === 0.5: round to nearest even integer
 * 6. Shift back by 10^decimals
 */
function bankerRound(value, decimals) {
  const factor = Math.pow(10, decimals);
  const shifted = value * factor;
  const floor = Math.floor(shifted);
  const remainder = shifted - floor;
  
  // Tolerance for floating-point comparison (±1e-10)
  const EPSILON = 1e-10;
  
  if (remainder < 0.5 - EPSILON) {
    // Round down
    return floor / factor;
  } else if (remainder > 0.5 + EPSILON) {
    // Round up
    return (floor + 1) / factor;
  } else {
    // Remainder ≈ 0.5: round to nearest even
    const rounded = floor % 2 === 0 ? floor : floor + 1;
    return rounded / factor;
  }
}
```

```javascript
function validateBillAmount(value) {
```

```javascript
  if (value === null || value === undefined || value === '') {

        message: 'Bill amount must be a number',
        constraint: 'typeof value === "number" or numeric string'
      }

      valid: false,
      error: {
        code: 'NEGATIVE_BILL_AMOUNT',
        message: 'Bill amount must be positive',
        constraint: 'value > 0'
      }

      valid: false,

        message: 'Bill amount must be at least 0.01',

      valid: false,

        code: 'BILL_AMOUNT_NOT_FINITE',
        message: 'Bill amount must be a finite number',
```

```javascript
  // Rule 5: Decimal places (at most 1)
  const decimalPlaces = (numValue.toString().split('.')[1] || '').length;
  if (decimalPlaces > 1) {
    return {
      valid: false,
      error: {
        code: 'TIP_PERCENTAGE_PRECISION',
        message: 'Tip percentage must have at most 1 decimal place',
        constraint: 'decimalPlaces <= 1'
      }
    };
  }

  // Rule 6: Not NaN or Infinity
  if (!Number.isFinite(numValue)) {
    return {
      valid: false,
      error: {
        code: 'TIP_PERCENTAGE_NOT_FINITE',
        message: 'Tip percentage must be a finite number',
        constraint: 'Number.isFinite(value)'
      }
    };
  }

  // All validations passed
  return { valid: true, error: null };
}
```

```javascript
  };
```

```javascript
function savePreferences(currencyCode) {
  // Step 1: Validate currency code
  if (!CURRENCY_METADATA[currencyCode]) {
    return {
      success: false,
      error: {
        code: 'INVALID_CURRENCY',
        message: `Currency ${currencyCode} not supported`
      }
    };
  }

  try {
    // Step 2: Create preference object
    const prefs = {
      selectedCurrency: currencyCode,
      lastUpdated: Date.now(),
      schemaVersion: 1
    };

    // Step 3: Serialize to JSON
    const json = JSON.stringify(prefs);

    // Step 4: Write to localStorage
    localStorage.setItem('tipCalculator_prefs_v1', json);

    // Step 5: Verify write (optional but recommended for financial apps)
    const verify = localStorage.getItem('tipCalculator_prefs_v1');
    if (verify !== json) {
      throw new Error('Verification failed: written data does not match');
    }

    // Step 6: Return success
    return { success: true, error: null };

  } catch (error) {
    // localStorage unavailable or quota exceeded
    let errorCode = 'STORAGE_ERROR';
    let errorMessage = 'Failed to save preferences';

    if (error.name === 'QuotaExceededError') {

      errorMessage = 'Storage quota exceeded; cannot save preferences';
    } else if (error.name === 'SecurityError') {
      errorCode = 'STORAGE_SECURITY_ERROR';
      errorMessage = 'Storage access denied (private browsing?)';

    console.warn(`${errorCode}: ${errorMessage}`, error);

    return {
      success: false,
      error: {
        code: errorCode,
        message: errorMessage
      }
    };
  }
}
```

### 7.1 Calculation Engine

#### 7.1.1 Core Calculation Function

**Function:** `calculateTip(billAmount, tipPercentage, currencyCode)`

**Pseudocode:**
```
FUNCTION calculateTip(billAmount, tipPercentage, currencyCode)
  // Step 1: Validate inputs
  IF NOT isValidBillAmount(billAmount) THEN
    THROW ValidationError("Invalid bill amount")
  END IF
  
  IF NOT isValidTipPercentage(tipPercentage) THEN
    THROW ValidationError("Invalid tip percentage")
  END IF
  
  IF NOT CURRENCY_METADATA[currencyCode] THEN
    THROW ValidationError("Invalid currency code")
  END IF
  
  // Step 2: Calculate with full precision (no rounding)
  tipAmountRaw ← billAmount × (tipPercentage ÷ 100)
  totalRaw ← billAmount + tipAmountRaw
  
  // Step 3: Get decimal places from currency metadata
  decimals ← CURRENCY_METADATA[currencyCode].decimals
  
  // Step 4: Apply banker's rounding
  tipAmountRounded ← bankerRound(tipAmountRaw, decimals)
  totalRounded ← bankerRound(totalRaw, decimals)
  
  // Step 5: Return result object
  RETURN {
    tipAmountRaw: tipAmountRaw,
    tipAmountRounded: tipAmountRounded,
    totalRaw: totalRaw,
    totalRounded: totalRounded,
    currencyCode: currencyCode,
    decimals: decimals,
    billAmount: billAmount,
    tipPercentage: tipPercentage
  }
END FUNCTION
```

**JavaScript Implementation:**
```javascript

  if (!isValidBillAmount(billAmount)) {
```

#### 7.1.2 Banker's Rounding Function

**Purpose:** Round monetary values to a specified number of decimal places using banker's rounding (round-half-to-even).

**Rationale:** Banker's rounding minimizes systematic bias in financial calculations. Standard round-half-up (0.5 always rounds up) introduces upward bias over many calculations. Banker's rounding rounds 0.5 to the nearest even number, distributing rounding error evenly.

**Example:**
- `bankerRound(2.5, 0)` → 2 (rounds to nearest even)
- `bankerRound(3.5, 0)` → 4 (rounds to nearest even)
- `bankerRound(2.25, 1)` → 2.2 (rounds to nearest even)
- `bankerRound(2.35, 1)` → 2.4 (rounds to nearest even)

**JavaScript Implementation:**
```javascript
function bankerRound(value, decimals) {
  // Validate inputs
  if (typeof value !== 'number' || typeof decimals !== 'number') {
    throw new TypeError('bankerRound requires numeric arguments');
  }
  if (decimals < 0 || !Number.isInteger(decimals)) {
    throw new RangeError('decimals must be a non-negative integer');
  }
  
  // Handle edge cases
  if (!isFinite(value)) {
    throw new RangeError('value must be a finite number');
  }
  
  // Special case: decimals = 0
  if (decimals === 0) {
    const floor = Math.floor(value);
    const remainder = value - floor;
    
    if (remainder < 0.5) return floor;
    if (remainder > 0.5) return floor + 1;
    // remainder === 0.5: round to nearest even
    return floor % 2 === 0 ? floor : floor + 1;
  }
  
  // General case: decimals > 0
  const factor = Math.pow(10, decimals);
  const shifted = value * factor;
  const floor = Math.floor(shifted);
  const remainder = shifted - floor;
  
  if (remainder < 0.5) return floor / factor;
  if (remainder > 0.5) return (floor + 1) / factor;
  // remainder === 0.5: round to nearest even
  return (floor % 2 === 0 ? floor : floor + 1) / factor;
}
```

**Validation Rules:**

| Rule | Condition | Error Message | Error Code |

### 7.3 Currency Formatting Engine

#### 7.3.1 Currency Formatter

**Function:** `CurrencyFormatter.format(value, currencyCode)`

**Purpose:** Format a numeric value as a localized currency string.

**Input:**
- `value` (number): The amount to format
- `currencyCode` (string): ISO 4217 currency code (e.g., 'USD', 'EUR', 'JPY')

**Output:** (string) Formatted currency string (e.g., '$50.00', '€50,00', '¥50')

**Implementation Using `Intl.NumberFormat`:**
```javascript
const CurrencyFormatter = {
  format(value, currencyCode) {
    // Validate inputs
    if (typeof value !== 'number' || !isFinite(value)) {
      throw new TypeError('value must be a finite number');
    }
    if (!CURRENCY_METADATA[currencyCode]) {
      throw new Error(`Unknown currency code: ${currencyCode}`);
    }
    
    // Get locale and decimal places from metadata
    const metadata = CURRENCY_METADATA[currencyCode];
    const locale = metadata.locale;
    const decimals = metadata.decimals;
    
    // Use Intl.NumberFormat for locale-aware formatting
    const formatter = new Intl.NumberFormat(locale, {
      style: 'currency',
      currency: currencyCode,
      minimumFractionDigits: decimals,
      maximumFractionDigits: decimals
    });
    
    return formatter.format(value);
  }
};
```

**Example Outputs:**
```javascript
CurrencyFormatter.format(50.00, 'USD')   // "$50.00"
CurrencyFormatter.format(50.00, 'EUR')   // "50,00 €" (locale-dependent)
CurrencyFormatter.format(50.00, 'GBP')   // "£50.00"
CurrencyFormatter.format(50.00, 'JPY')   // "¥50"
CurrencyFormatter.format(50.00, 'INR')   // "₹50.00"
CurrencyFormatter.format(1234.56, 'USD') // "$1,234.56"
CurrencyFormatter.format(1234.56, 'EUR') // "1.234,56 €" (locale-dependent)
```

**Constraints:**
- Must use `Intl.NumberFormat` for locale-aware formatting (not manual string concatenation)
- Must respect decimal places from `CURRENCY_METADATA` (JPY = 0, others = 2)
- Must handle thousands separators correctly per locale
- Must handle negative values (e.g., "-$50.00")
- Must be deterministic (same input always produces same output)

**Browser Compatibility:**
- `Intl.NumberFormat` is supported in all modern browsers (IE 11+)
- Fallback for older browsers: manual formatting (not required for MVP)

### 7.6 Error Handling Strategy

#### 7.6.1 Error Types & Recovery

| Error Type | Cause | Recovery | User Feedback |
|---|---|---|---|
| **ValidationError** | Invalid user input (non-numeric, out of range) | Clear results; show inline error message | "Please enter a valid bill amount" |
| **CalculationError** | Unexpected error in calculation logic | Log error; clear results | "An error occurred. Please try again." |
| **StorageError** | `localStorage` unavailable (quota exceeded, private browsing) | Continue without persistence | None (graceful degradation) |
| **FormattingError** | Invalid currency code or formatting failure | Use fallback format (e.g., "50.00 USD") | None (fallback display) |

**Error Handling Pattern:**
```javascript
try {
  // Attempt operation
  const result = calculateTip(billAmount, tipPercentage, currencyCode);

  // Log error

  // Clear results
  clearResultsDisplay();
}
```

**Constraints:**
- All errors must be caught and handled gracefully (no unhandled exceptions)
- User-facing error messages must be friendly and actionable
- Technical errors must be logged to console for debugging
- App must remain functional after error (no crashes or frozen state)

### 7.2 Module: `calculator.js`

**Purpose:** Pure calculation functions. No side effects, no DOM access, no I/O.

**Exports:**

```javascript
/**
 * calculator.js - Pure calculation engine
 * 
 * All functions are deterministic and side-effect-free.
 * Suitable for unit testing and server-side replication (if backend added).
 */

/**
 * Banker's rounding implementation (round-half-to-even)
 * 
 * @param {number} value - Value to round
 * @param {number} decimals - Number of decimal places
 * @returns {number} Rounded value
 * 
 * Algorithm:
 * - If fractional part < 0.5: round down
 * - If fractional part > 0.5: round up
 * - If fractional part === 0.5: round to nearest even number
 * 
 * Example:
 *   bankerRound(2.5, 0) => 2 (2 is even)
 *   bankerRound(3.5, 0) => 4 (4 is even)
 *   bankerRound(2.25, 1) => 2.2 (2 is even in second decimal)
 */
export function bankerRound(value, decimals) {
  if (typeof value !== 'number' || typeof decimals !== 'number') {
    throw new TypeError('bankerRound: value and decimals must be numbers');
  }
  if (decimals < 0 || !Number.isInteger(decimals)) {
    throw new RangeError('bankerRound: decimals must be non-negative integer');
  }
  if (!Number.isFinite(value)) {
    throw new RangeError('bankerRound: value must be finite');
  }

  const factor = Math.pow(10, decimals);
  const shifted = value * factor;
  const floor = Math.floor(shifted);
  const remainder = shifted - floor;

  // Tolerance for floating-point comparison (IEEE 754)
  const EPSILON = 1e-10;

  if (remainder < 0.5 - EPSILON) {
    return floor / factor;
  }
  if (remainder > 0.5 + EPSILON) {
    return (floor + 1) / factor;
  }
  // remainder ≈ 0.5: round to nearest even
  return (floor % 2 === 0 ? floor : floor + 1) / factor;
}

/**
 * Core tip calculation function
 * 
 * @param {number} billAmount - Bill amount (0.01 to 999,999.99)
 * @param {number} tipPercentage - Tip percentage (0 to 100)
 * @param {string} currencyCode - ISO 4217 code (e.g., 'USD', 'EUR')
 * @returns {CalculationResult}
 * 
 * @throws {TypeError} If inputs are not numbers or currencyCode is not string
 * @throws {RangeError} If inputs are out of valid range
 * @throws {Error} If currencyCode is not recognized
 * 
 * CalculationResult = {
 *   tipAmountRaw: number,           // Full precision (no rounding)
 *   tipAmountRounded: number,       // Banker's rounded to currency decimals
 *   totalRaw: number,               // bill + tipAmountRaw (full precision)
 *   totalRounded: number,           // Banker's rounded to currency decimals
 *   currencyCode: string,
 *   decimals: number,               // Decimal places for this currency
 *   calculatedAt: number            // ISO 8601 timestamp (ms)
 * }
 */
export function calculateTip(billAmount, tipPercentage, currencyCode) {
  // Input type validation
  if (typeof billAmount !== 'number' || typeof tipPercentage !== 'number') {
    throw new TypeError('calculateTip: billAmount and tipPercentage must be numbers');
  }
  if (typeof currencyCode !== 'string') {
    throw new TypeError('calculateTip: currencyCode must be string');
  }

  // Input range validation
  if (!Number.isFinite(billAmount)) {
    throw new RangeError('calculateTip: billAmount must be finite');
  }
  if (!Number.isFinite(tipPercentage)) {
    throw new RangeError('calculateTip: tipPercentage must be finite');
  }
  if (billAmount < 0.01 || billAmount > 999999.99) {
    throw new RangeError('calculateTip: billAmount must be between 0.01 and 999,999.99');
  }
  if (tipPercentage < 0 || tipPercentage > 100) {
    throw new RangeError('calculateTip: tipPercentage must be between 0 and 100');
  }

  // Currency validation (will be imported from constants.js)
  const currencyMetadata = CURRENCY_METADATA[currencyCode];
  if (!currencyMetadata) {
    throw new Error(`calculateTip: Unknown currency code '${currencyCode}'`);
  }

  // Step 1: Calculate with full precision (no rounding)
  const tipAmountRaw = billAmount * (tipPercentage / 100);
  const totalRaw = billAmount + tipAmountRaw;

  // Step 2: Apply banker's rounding to currency precision
  const decimals = currencyMetadata.decimals;
  const tipAmountRounded = bankerRound(tipAmountRaw, decimals);
  const totalRounded = bankerRound(totalRaw, decimals);

  // Step 3: Return calculation result with audit trail
  return {
    tipAmountRaw,
    tipAmountRounded,
    totalRaw,
    totalRounded,
    currencyCode,
    decimals,
    calculatedAt: Date.now()
  };
}

/**
 * Recalculate with new currency (re-round existing raw values)
 * 
 * Used when user changes currency without changing bill/tip inputs.
 * Avoids recalculating from scratch; just re-rounds to new currency's decimals.
 * 
 * @param {CalculationResult} previousResult - Result from calculateTip()
 * @param {string} newCurrencyCode - New ISO 4217 code
 * @returns {CalculationResult} New result with same raw values, new rounding
 */
export function recalculateForCurrency(previousResult, newCurrencyCode) {
  if (!previousResult || typeof previousResult !== 'object') {
    throw new TypeError('recalculateForCurrency: previousResult must be object');
  }
  if (typeof newCurrencyCode !== 'string') {
    throw new TypeError('recalculateForCurrency: newCurrencyCode must be string');
  }

  const newCurrencyMetadata = CURRENCY_METADATA[newCurrencyCode];
  if (!newCurrencyMetadata) {
    throw new Error(`recalculateForCurrency: Unknown currency code '${newCurrencyCode}'`);
  }

  const newDecimals = newCurrencyMetadata.decimals;
  return {
    tipAmountRaw: previousResult.tipAmountRaw,
    tipAmountRounded: bankerRound(previousResult.tipAmountRaw, newDecimals),
    totalRaw: previousResult.totalRaw,
    totalRounded: bankerRound(previousResult.totalRaw, newDecimals),
    currencyCode: newCurrencyCode,
    decimals: newDecimals,
    calculatedAt: Date.now()
  };
}
```

```javascript
      isValid: false,
      value: null,
      error: 'Please enter a bill amount',

      isValid: false,
      value: null,
      error: 'Bill amount must be a number',
      errorCode: 'BILL_AMOUNT_NOT_NUMERIC'
    };

    return {
      isValid: false,
      value: null,
      error: 'Bill amount cannot be negative',
      errorCode: 'BILL_AMOUNT_NEGATIVE'

      value: null,

      isValid: false,
      value: null,
      error: 'Bill amount cannot exceed $999,999.99',
      errorCode: 'BILL_AMOUNT_TOO_LARGE'

      isValid: false,
      value: null,
```

```javascript
    return true;

/**
 * Load user preferences from localStorage
 * 
 * @returns {StorageSchema | null} Preferences object, or null if not found or invalid
 * 
 * Behavior:
 * - If localStorage unavailable, returns null (graceful degradation)
 * - If key not found, returns null
 * - If data is corrupted (invalid JSON), returns null and logs warning
 * - If schema version mismatch, returns null (for future migrations)
 * - If currency code invalid, returns null (data corruption recovery)
 */
export function loadPreferences() {
  if (!isStorageAvailable()) {
    console.warn('storage.js: localStorage unavailable; preferences will not persist');
    return null;
  }

  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (!stored) {
      return null; // Key not found
    }

    const parsed = JSON.parse(stored);

    // Validate schema version
    if (parsed.schemaVersion !== 1) {
      console.warn(`storage.js: Unknown schema version ${parsed.schemaVersion}; ignoring stored preferences`);
      return null;
    }

    // Validate currency code
    if (!CURRENCY_METADATA[parsed.selectedCurrency]) {
      console.warn(`storage.js: Invalid currency code '${parsed.selectedCurrency}' in stored preferences; ignoring`);
      return null;
    }

    // Validate timestamp
    if (typeof parsed.lastUpdated !== 'number' || parsed.lastUpdated > Date.now()) {
      console.warn('storage.js: Invalid timestamp in stored preferences; ignoring');
      return null;
    }

    return parsed;
  } catch (e) {
    console.error('storage.js: Error parsing stored preferences:', e);
    return null;
  }
}

/**
 * Save user preferences to localStorage
 * 
 * @param {string} currencyCode - ISO 4217 code to save
 * @returns {boolean} True if save successful, false otherwise
 * 
 * Behavior:
 * - If localStorage unavailable, returns false (graceful degradation)
 * - If currencyCode invalid, throws TypeError
 * - If save fails (quota exceeded), logs error and returns false
 * - Always includes schemaVersion for future migrations
 */
export function savePreferences(currencyCode) {
  if (typeof currencyCode !== 'string') {
    throw new TypeError('savePreferences: currencyCode must be string');
  }

  if (!CURRENCY_METADATA[currencyCode]) {
    throw new Error(`savePreferences: Unknown currency code '${currencyCode}'`);
  }

  if (!isStorageAvailable()) {
    console.warn('storage.js: localStorage unavailable; preferences will not persist');
    return false;
  }

  try {
    const prefs = {
      selectedCurrency: currencyCode,
      lastUpdated: Date.now(),
      schemaVersion: 1
    };

    localStorage.setItem(STORAGE_KEY, JSON.stringify(prefs));
    return true;
  } catch (e) {
    if (e.name === 'QuotaExceededError') {
      console.error('storage.js: localStorage quota exceeded; cannot save preferences');
    } else {
      console.error('storage.js: Error saving preferences:', e);
    }
    return false;
  }
}

/**

 *

 */
export function clearPreferences() {
  if (!isStorageAvailable()) {
    return false;
  }

  try {
    localStorage.removeItem(STORAGE_KEY);
    return true;
  } catch (e) {
    console.error('storage.js: Error clearing preferences:', e);
    return false;
  }
}
```

### 8.1 Error Handling Strategy

**Philosophy:** Fail gracefully. Never crash the app; always provide user-friendly feedback.

**Error Categories:**

| **Storage Errors** | localStorage unavailable | Continue without persistence | None (silent degradation) |
| **Browser Compatibility** | Missing APIs (Intl.NumberFormat) | Fallback to basic formatting | None (app still works) |

## 10. VALIDATION RULES

### 10.1 Input Validation Matrix

| Field | Type | Min | Max | Decimals | Empty OK? | Validation Function |
|---|---|---|---|---|---|---|
| Bill Amount | number | 0.01 | 999,999.99 | 2 | No | `validateBillAmount()` |
| Tip Percentage | number | 0 | 100 | 1 | Yes (→ 0%) | `validateTipPercentage()` |
| Currency Code | string | N/A | N/A | N/A | No | `validateCurrencyCode()` |

---

### 10.2 Validation Rules (Detailed)

**Bill Amount Validation:**

```javascript
// Rule 1: Must not be empty
if (input === '') → INVALID (BILL_AMOUNT_EMPTY)

// Rule 2: Must be numeric
if (isNaN(parseFloat(input))) → INVALID (BILL_AMOUNT_NOT_NUMERIC)

// Rule 3: Must not be negative
if (parseFloat(input) < 0) → INVALID (BILL_AMOUNT_NEGATIVE)

// Rule 4: Must be >= 0.01
if (parseFloat(input) < 0.01) → INVALID (BILL_AMOUNT_TOO_SMALL)

// Rule 5: Must be <= 999,999.99
if (parseFloat(input) > 999999.99) → INVALID (BILL_AMOUNT_TOO_LARGE)

// Rule 6: Must have at most 2 decimal places
if (input.match(/\.(\d+)$/) && match[1].length > 2) → INVALID (BILL_AMOUNT_TOO_MANY_DECIMALS)

// All rules pass → VALID
```

**Tip Percentage Validation:**

```javascript
// Rule 1: Empty is OK (treat as 0%)
if (input === '') → VALID (value = 0)

// Rule 2: Must be numeric
if (input !== '' && isNaN(parseFloat(input))) → INVALID (TIP_PERCENTAGE_NOT_NUMERIC)

// Rule 3: Must not be negative
if (parseFloat(input) < 0) → INVALID (TIP_PERCENTAGE_NEGATIVE)

// Rule 4: Must be <= 100
if (parseFloat(input) > 100) → INVALID (TIP_PERCENTAGE_TOO_LARGE)

// Rule 5: Must have at most 1 decimal place
if (input.match(/\.(\d+)$/) && match[1].length > 1) → INVALID (TIP_PERCENTAGE_TOO_MANY_DECIMALS)

// All rules pass → VALID
```

**Currency Code Validation:**

```javascript
// Rule 1: Must be string
if (typeof input !== 'string') → INVALID (CURRENCY_CODE_NOT_STRING)

// Rule 2: Must be exactly 3 uppercase letters
if (!/^[A-Z]{3}$/.test(input)) → INVALID (CURRENCY_CODE_INVALID_FORMAT)

// Rule 3: Must exist in CURRENCY_METADATA
if (!CURRENCY_METADATA[input]) → INVALID (CURRENCY_CODE_NOT_SUPPORTED)

// All rules pass → VALID
```

### 10.3 Validation Timing

**Real-Time Validation (on input change):**
- Bill amount: Validate after each keystroke (debounced 100ms)
- Tip percentage: Validate after each keystroke (debounced 100ms)
- Currency selector: Validate on change (no debounce needed)

**Validation on Submit (if "Calculate" button added in future):**

### 14.4 Security Headers (if using web server)

**Recommended HTTP Headers:**

```
Content-Security-Policy: default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline'
X-Content-Type-Options: nosniff
X-Frame-Options: DENY
X-XSS-Protection: 1; mode=block
Referrer-Policy: no-referrer
Permissions-Policy: geolocation=(), microphone=(), camera=()
```

**Rationale:**
- `Content-Security-Policy`: Prevent XSS attacks; allow inline scripts (single-file app)
- `X-Content-Type-Options`: Prevent MIME type sniffing
- `X-Frame-Options`: Prevent clickjacking
- `X-XSS-Protection`: Legacy XSS protection (for older browsers)
- `Referrer-Policy`: Don't leak referrer information
- `Permissions-Policy`: Disable unnecessary browser features
